/** * Ngapori RAMS — Field Maintenance Portal * Apps Script backend for index.html. * * SETUP * 1. Open Ngapori_RAMS_v2 in Google Sheets. * 2. Extensions > Apps Script. * 3. If it IS bound (the normal setup), leave it blank — * SpreadsheetApp.getActiveSpreadsheet() will find it automatically. */ const SHEET_ID = ''; function getSpreadsheet_() { return SHEET_ID ? SpreadsheetApp.openById(SHEET_ID) : SpreadsheetApp.getActiveSpreadsheet(); } /** * Diagnostic: reports exactly which spreadsheet this deployment is writing to, * plus a live row count for ASSET_REGISTRY. Call this from the app (or manually * from the Apps Script editor via Run) any time you're unsure whether you're * looking at the same file the web app is actually connected to. */ function getBackendInfo() { const ss = getSpreadsheet_(); const sheet = ss.getSheetByName('ASSET_REGISTRY'); const lastRow = sheet ? sheet.getLastRow() : -1; const assetCount = sheet ? sheet.getRange(14, 1, Math.max(lastRow - 13, 0), 1).getValues() .filter(function (r) { return r[0] !== '' && r[0] !== null; }).length : -1; return { spreadsheetName: ss.getName(), spreadsheetId: ss.getId(), spreadsheetUrl: ss.getUrl(), assetRegistryRowCount: assetCount, scriptTimeZone: Session.getScriptTimeZone(), executedAt: new Date().toISOString() }; } function doGet(e) { return HtmlService.createHtmlOutputFromFile('index') .setTitle('Ngapori RAMS — Field Maintenance Portal') .addMetaTag('viewport', 'width=device-width, initial-scale=1') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } /** Snapshot of USERS, ASSET_REGISTRY, PARTS_INVENTORY and SETTINGS for the app to boot from. */ function getInitialData() { const ss = getSpreadsheet_(); const users = readTable_(ss.getSheetByName('USERS'), 5, ['username', 'pin', 'displayName', 'role', 'active']) .filter(function (u) { return u.username && String(u.active).toLowerCase() === 'yes'; }) .map(function (u) { return { username: u.username, pin: String(u.pin), displayName: u.displayName, role: u.role }; }); const assets = readTable_(ss.getSheetByName('ASSET_REGISTRY'), 13, ['assetId', 'assetName', 'category', 'location', 'status', 'serviceIntervalDays', 'lastServiced', 'nextServiceDue', 'purchasePrice', 'qrCodeUrl', 'identifierType', 'identifierNumber', 'photoUrls', 'manufacturer', 'dateCreated', 'assetType']) .filter(function (a) { return a.assetId; }) .map(function (a) { a.lastServiced = fmtDate_(a.lastServiced); a.nextServiceDue = fmtDate_(a.nextServiceDue); a.dateCreated = fmtDate_(a.dateCreated); return a; }); const parts = readTable_(ss.getSheetByName('PARTS_INVENTORY'), 5, ['name', 'onHand', 'minimum', 'status']) .filter(function (p) { return p.name; }); const assetTypes = readAssetTypes_(ss); const domainOptions = []; assetTypes.forEach(function (t) { if (domainOptions.indexOf(t.domain) === -1) domainOptions.push(t.domain); }); const settingsSheet = ss.getSheetByName('SETTINGS'); const settings = { statusOptions: readColumn_(settingsSheet, 6, 8, 1), domainOptions: domainOptions, assetTypes: assetTypes, workTypes: readColumn_(settingsSheet, 6, 9, 3), roles: readColumn_(settingsSheet, 6, 8, 4), defaultServiceIntervalDays: settingsSheet.getRange('B14').getValue(), lowStockMultiplier: settingsSheet.getRange('B15').getValue(), overdueGraceDays: settingsSheet.getRange('B16').getValue() }; return { users: users, assets: assets, parts: parts, settings: settings }; } /** Checks a username/PIN pair against USERS. Returns the user (minus PIN) or null. */ function authenticate(username, pin) { const ss = getSpreadsheet_(); const users = readTable_(ss.getSheetByName('USERS'), 5, ['username', 'pin', 'displayName', 'role', 'active']); const match = users.find(function (u) { return u.username === username && String(u.pin) === String(pin) && String(u.active).toLowerCase() === 'yes'; }); return match ? { username: match.username, displayName: match.displayName, role: match.role } : null; } /** * Appends one work order to MAINTENANCE_LOG, deducts any parts used from PARTS_INVENTORY, * and — if the worker flagged it — flips the asset to "Out of Order" in ASSET_REGISTRY. * Returns the new Log_ID plus a fresh data snapshot so the UI can re-render KPIs. * * entry: { assetId, workType, description, workPerformed, reportedBy, role, assignedTo, * priority ("Low"|"Medium"|"High"|"Emergency"), cost, hours, completedDate, * status, parts: [{name, quantity}], materialsNote, markOutOfOrder, serviceLocation, * vendorName, managerSignoff } */ function submitLogEntry(entry) { const ss = getSpreadsheet_(); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const logSheet = ss.getSheetByName('MAINTENANCE_LOG'); ensureMaintenanceLogColumns_(logSheet); const logId = nextLogId_(logSheet); const timestamp = new Date(); logSheet.appendRow([ logId, timestamp, entry.assetId, entry.workType, entry.description, entry.reportedBy, entry.role, Number(entry.cost) || 0, entry.status, entry.managerSignoff || '', entry.serviceLocation || 'Ranch Crew', entry.vendorName || '', entry.priority || 'Medium', entry.assignedTo || entry.reportedBy, Number(entry.hours) || 0, entry.completedDate ? new Date(entry.completedDate) : '', entry.workPerformed || '', entry.materialsNote || '' ]); (entry.parts || []).forEach(function (p) { deductPart_(ss.getSheetByName('PARTS_INVENTORY'), p.name, Number(p.quantity) || 0); }); if (entry.markOutOfOrder) { setAssetStatus_(ss.getSheetByName('ASSET_REGISTRY'), entry.assetId, 'Out of Order'); } SpreadsheetApp.flush(); return { logId: logId, timestamp: timestamp.toISOString(), data: getInitialData() }; } finally { lock.releaseLock(); } } /** Adds the Service_Location / Vendor_Name / Priority / Assigned_To / Hours / Completed_Date / Work_Performed / Materials_Note header columns (K–R) to MAINTENANCE_LOG if missing. */ function ensureMaintenanceLogColumns_(sheet) { var headers = { K5: 'Service_Location', L5: 'Vendor_Name', M5: 'Priority', N5: 'Assigned_To', O5: 'Hours', P5: 'Completed_Date', Q5: 'Work_Performed', R5: 'Materials_Note' }; var cellRefs = Object.keys(headers); var ranges = cellRefs.map(function (ref) { return sheet.getRange(ref); }); var currentValues = ranges.map(function (cell) { return cell.getValue(); }); cellRefs.forEach(function (cellRef, i) { if (currentValues[i] !== headers[cellRef]) ranges[i].setValue(headers[cellRef]); }); } /** * Registers a brand-new asset into ASSET_REGISTRY. Fills the first blank row in the * pre-built 14–300 range (so it inherits that row's existing Next_Service_Due and * QR_Code_URL formulas instead of overwriting them) and adds Identifier_Type / * Identifier_Number columns to the sheet the first time it's called. * * data: { assetId, assetName, category (domain, e.g. "Equipment"), assetType (specific * type, e.g. "Tractor" — looked up against ASSET_TYPES for Auto-Generate ID), * location, status, serviceIntervalDays, lastServiced, purchasePrice, * identifierType, identifierNumber, manufacturer, * photos: [{filename, mimeType, base64}] } — photos requires at least 3, enforced client-side. * Date_Created is stamped server-side, not taken from the client. */ function registerAsset(data) { const ss = getSpreadsheet_(); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const sheet = ss.getSheetByName('ASSET_REGISTRY'); if (!sheet) { throw new Error('ASSET_REGISTRY tab not found in spreadsheet "' + ss.getName() + '" (' + ss.getUrl() + '). This deployment is not writing to the sheet you expect — check SHEET_ID / which file this script is bound to.'); } ensureAssetRegistryColumns_(sheet); const assetId = String(data.assetId || '').trim(); if (!assetId) throw new Error('Ranch Asset ID is required.'); const lastRow = Math.max(sheet.getLastRow(), 300); const ids = sheet.getRange(14, 1, lastRow - 13, 1).getValues(); var targetRow = -1; for (var i = 0; i < ids.length; i++) { var existing = String(ids[i][0] || '').trim(); if (existing && existing.toLowerCase() === assetId.toLowerCase()) { throw new Error('Asset ID "' + assetId + '" already exists in ASSET_REGISTRY.'); } if (!existing && targetRow === -1) targetRow = 14 + i; } if (targetRow === -1) { targetRow = lastRow + 1; // Extend formulas into the new row so Next_Service_Due / QR_Code_URL still compute. sheet.getRange('H300').copyTo(sheet.getRange('H' + targetRow)); sheet.getRange('J300').copyTo(sheet.getRange('J' + targetRow)); } sheet.getRange(targetRow, 1, 1, 6).setValues([[ assetId, data.assetName || '', data.category || '', data.location || '', data.status || 'Operational', Number(data.serviceIntervalDays) || 90 ]]); sheet.getRange(targetRow, 7).setValue(data.lastServiced ? new Date(data.lastServiced) : ''); sheet.getRange(targetRow, 9).setValue(Number(data.purchasePrice) || 0); sheet.getRange(targetRow, 11, 1, 2).setValues([[data.identifierType || '', data.identifierNumber || '']]); sheet.getRange(targetRow, 14, 1, 2).setValues([[data.manufacturer || '', new Date()]]); sheet.getRange(targetRow, 16).setValue(data.assetType || ''); const photoUrls = uploadAssetPhotos_(assetId, data.photos); if (photoUrls.length) sheet.getRange(targetRow, 13).setValue(photoUrls.join(', ')); SpreadsheetApp.flush(); return { assetId: assetId, data: getInitialData() }; } finally { lock.releaseLock(); } } /** * Suggests the next Ranch Asset ID for a specific asset type (e.g. "Tractor", "Barn", * "Checkpoint") by looking up its prefix + suffix style in the ASSET_TYPES tab, then * scanning ASSET_REGISTRY for IDs already using that prefix. * - "numeric" types (most equipment/infrastructure) increment -01, -02, -03… * - "alpha" types (named/lettered instances like Barn or Checkpoint) increment -A, -B, -C… * Purely a suggestion — the field stays editable. */ function suggestNextAssetId(assetType) { const ss = getSpreadsheet_(); const typeInfo = findAssetType_(ss, assetType); const prefix = typeInfo ? typeInfo.prefix : 'AST'; const suffixStyle = typeInfo ? typeInfo.suffixStyle : 'numeric'; const sheet = ss.getSheetByName('ASSET_REGISTRY'); const lastRow = sheet.getLastRow(); const existingSuffixes = []; if (lastRow >= 14) { const ids = sheet.getRange(14, 1, lastRow - 13, 1).getValues(); ids.forEach(function (row) { const id = String(row[0] || '').trim(); if (!id) return; if (id.toUpperCase().indexOf(prefix.toUpperCase() + '-') === 0) { existingSuffixes.push(id.substring(prefix.length + 1)); } }); } if (suffixStyle === 'alpha') { var maxLetterCode = 'A'.charCodeAt(0) - 1; existingSuffixes.forEach(function (s) { const m = s.match(/^([A-Za-z]+)$/); if (m && m[1].length === 1) { const code = m[1].toUpperCase().charCodeAt(0); if (code > maxLetterCode) maxLetterCode = code; } }); return prefix + '-' + String.fromCharCode(maxLetterCode + 1); } var maxNum = 0; existingSuffixes.forEach(function (s) { const n = parseInt(s, 10); if (!isNaN(n) && n > maxNum) maxNum = n; }); const next = maxNum + 1; return prefix + '-' + (next < 10 ? '0' + next : String(next)); } /** Reads the ASSET_TYPES tab (Domain, Asset_Type, Prefix, Suffix_Style, Workbook_Mapping) starting at row 4. */ function readAssetTypes_(ss) { const sheet = ss.getSheetByName('ASSET_TYPES'); if (!sheet) return []; const lastRow = sheet.getLastRow(); if (lastRow < 4) return []; return sheet.getRange(4, 1, lastRow - 3, 4).getValues() .filter(function (row) { return row[0] && row[1]; }) .map(function (row) { return { domain: row[0], type: row[1], prefix: row[2], suffixStyle: (row[3] || 'numeric').toLowerCase() }; }); } function findAssetType_(ss, assetType) { const types = readAssetTypes_(ss); return types.find(function (t) { return t.type === assetType; }) || null; } /** Adds the Identifier_Type / Identifier_Number / Photo_URLs / Manufacturer / Date_Created / Asset_Type header columns (K–P) if missing. */ function ensureAssetRegistryColumns_(sheet) { var headers = { K13: 'Identifier_Type', L13: 'Identifier_Number', M13: 'Photo_URLs', N13: 'Manufacturer', O13: 'Date_Created', P13: 'Asset_Type' }; var cellRefs = Object.keys(headers); var ranges = cellRefs.map(function (ref) { return sheet.getRange(ref); }); var currentValues = ranges.map(function (cell) { return cell.getValue(); }); cellRefs.forEach(function (cellRef, i) { if (currentValues[i] !== headers[cellRef]) ranges[i].setValue(headers[cellRef]); }); } /** * Uploads base64-encoded photos to Drive under "RAMS Asset Photos//", shares each * file as view-only via link, and returns the resulting URLs. photos: [{filename, mimeType, base64}] */ function uploadAssetPhotos_(assetId, photos) { if (!photos || !photos.length) return []; const rootFolders = DriveApp.getFoldersByName('RAMS Asset Photos'); const root = rootFolders.hasNext() ? rootFolders.next() : DriveApp.createFolder('RAMS Asset Photos'); const existing = root.getFoldersByName(assetId); const folder = existing.hasNext() ? existing.next() : root.createFolder(assetId); return photos.map(function (p, i) { const bytes = Utilities.base64Decode(p.base64); const name = p.filename || (assetId + '-photo-' + (i + 1) + '.jpg'); const blob = Utilities.newBlob(bytes, p.mimeType || 'image/jpeg', name); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return file.getUrl(); }); } // --------------------------------------------------------------------------- // helpers // --------------------------------------------------------------------------- function readTable_(sheet, headerRow, keys) { const lastRow = sheet.getLastRow(); if (lastRow <= headerRow) return []; const range = sheet.getRange(headerRow + 1, 1, lastRow - headerRow, keys.length); return range.getValues() .filter(function (row) { return row.some(function (v) { return v !== '' && v !== null; }); }) .map(function (row) { const obj = {}; keys.forEach(function (k, i) { obj[k] = row[i]; }); return obj; }); } function readColumn_(sheet, startRow, endRow, col) { return sheet.getRange(startRow, col, endRow - startRow + 1, 1).getValues() .map(function (r) { return r[0]; }) .filter(function (v) { return v !== '' && v !== null; }); } function fmtDate_(v) { if (!v) return ''; if (Object.prototype.toString.call(v) === '[object Date]') { return Utilities.formatDate(v, Session.getScriptTimeZone() || 'UTC', 'yyyy-MM-dd'); } return String(v); } function nextLogId_(logSheet) { const lastRow = logSheet.getLastRow(); const ids = lastRow > 5 ? logSheet.getRange(6, 1, lastRow - 5, 1).getValues().map(function (r) { return r[0]; }) : []; const nums = ids.map(function (id) { return parseInt(String(id).replace('LOG-', ''), 10); }).filter(function (n) { return !isNaN(n); }); const next = (nums.length ? Math.max.apply(null, nums) : 1000) + 1; return 'LOG-' + next; } function deductPart_(partsSheet, partName, qty) { const lastRow = partsSheet.getLastRow(); if (lastRow < 6) return; const names = partsSheet.getRange(6, 1, lastRow - 5, 1).getValues(); for (let i = 0; i < names.length; i++) { if (names[i][0] === partName) { const cell = partsSheet.getRange(6 + i, 2); cell.setValue(Math.max(0, Number(cell.getValue()) - qty)); break; } } } function setAssetStatus_(assetSheet, assetId, status) { const lastRow = assetSheet.getLastRow(); if (lastRow < 14) return; const ids = assetSheet.getRange(14, 1, lastRow - 13, 1).getValues(); for (let i = 0; i < ids.length; i++) { if (ids[i][0] === assetId) { assetSheet.getRange(14 + i, 5).setValue(status); break; } } }
